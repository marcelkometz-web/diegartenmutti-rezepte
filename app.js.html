async function loadRecipesFromSupabase(){
  const status = document.getElementById('syncStatus'); status.innerText = '⏳ Lade...';
  try {
    const { data, error } = await supabaseClient.from('Rezepte').select('*').order('id', { ascending: true });
    if (error) throw error;
    recipes = data.map(item => ({
      id: item.id, title: item.title || 'Rezept', emoji: item.emoji || '🍲',
      category: Array.isArray(item.category)?item.category:[item.category],
      airfryer: item.airfryer===true, score: item.score||'A',
      scoreClass: 'var(--score-a)', selectedSide: item.selectedSide||'keine',
      ingredients: item.ingredients? item.ingredients.split(',') : [],
      zutatenListe: item.zutaten_liste||'', tipps: item.tipps||'', docCheck: item.doc_check||'',
      image_url: item.image_url||null,
      baseNutri: { kcal: Number(item.kcal)||300, kh: Number(item.kh)||5, sugar: Number(item.sugar)||2, fat: Number(item.fat)||10, prot: Number(item.prot)||25 },
      prep: item.prep||'', stoveStep: item.stove_step||'', ovenStep: item.oven_step||'', airfryerStep: item.airfryer_step||'',
      mealType: (item.title||'').toLowerCase().includes('frühstück')?'fruehstueck':'mittag'
    }));
    status.innerText = `⚡ ${recipes.length} Rezepte synchronisiert`;
    filterRecipes();
  } catch(e){ document.getElementById('syncStatus').innerText='⚡ Offline'; }
}

function filterRecipes(){
  const val = document.getElementById('searchInput').value.toLowerCase().trim();
  const terms = val ? val.split(/\s+/) : [];
  const filtered = recipes.filter(r=>{
    const matches = terms.length===0 || terms.every(t=> r.title.toLowerCase().includes(t));
    const catMatch = currentCategory==='all' || r.mealType===currentCategory;
    return matches && catMatch;
  });
  const c = document.getElementById('recipeList');
  c.innerHTML = filtered.map(r=> `<div class="recipe-card" onclick="openDetails('${r.id}')"><div class="card-icon">${r.image_url?`<img src="${r.image_url}" style="width:100%;height:100%;object-fit:cover;border-radius:12px">`:r.emoji}</div><div class="card-meta"><h3>${escapeHtml(r.title)}</h3></div><div class="card-quick-macros">🔥 ${r.baseNutri.kcal} kcal • 💪 ${r.baseNutri.prot}g</div></div>`).join('') || '<div style="grid-column:1/-1;text-align:center;padding:40px;color:var(--text-muted)">Keine Rezepte gefunden</div>';
}
