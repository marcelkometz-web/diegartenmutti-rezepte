async function saveNewRecipe() { const title = document.getElementById('newTitle').value.trim(); const emoji = document.getElementById('newEmoji').value.trim() || '🍲'; const score = document.getElementById('newScore').value || 'A'; const mealCat = document.getElementById('newMealCat').value; const kcal = Number(document.getElementById('newKcal').value) || 300; const prot = Number(document.getElementById('newProt').value) || 25; const kh = Number(document.getElementById('newKh').value) || 5; const sugar = Number(document.getElementById('newSugar').value) || 2; const fat = Number(document.getElementById('newFat').value) || 10; const airfryer = document.getElementById('newAirfryer').value === 'true'; const ingredients = document.getElementById('newIngredients').value.trim(); const zutaten = document.getElementById('newZutaten').value.trim(); const prep = document.getElementById('newPrep').value.trim() || 'Zutaten vorbereiten.'; const stove = document.getElementById('newStove').value.trim(); const oven = document.getElementById('newOven').value.trim(); const afStep = document.getElementById('newAirfryerStep').value.trim(); const docCheck = document.getElementById('newDocCheck').value.trim(); const tipps = document.getElementById('newTipps').value.trim(); if (!title) { alert('Bitte gib einen Titel für das Rezept ein.'); return; } const categories = [mealCat, 'lowcarb']; if (airfryer) categories.push('airfryer'); const saveBtn = document.querySelector('.btn-save'); if (saveBtn) { saveBtn.innerText = '⏳ Speichere in Datenbank...'; saveBtn.disabled = true; } const { error } = await supabaseClient.from('Rezepte').insert([{ title: title, emoji: emoji, score: score, category: categories, airfryer: airfryer, kcal: kcal, prot: prot, kh: kh, sugar: sugar, fat: fat, ingredients: ingredients, zutaten_liste: zutaten, prep: prep, stove_step: stove, oven_step: oven, airfryer_step: afStep, doc_check: docCheck, tipps: tipps }]); if (!error) { if (saveBtn) saveBtn.innerText = '✅ Erfolgreich gespeichert!'; setTimeout(() => { closeSheetForce(); triggerRefresh(); }, 600); } else { alert('Fehler beim Speichern in Supabase: ' + error.message); if (saveBtn) { saveBtn.innerText = '💾 Rezept in Datenbank speichern'; saveBtn.disabled = false; } } }
